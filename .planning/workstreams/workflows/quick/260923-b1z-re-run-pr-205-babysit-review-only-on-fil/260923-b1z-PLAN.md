scope_endpoint=27678153

sha256sum .claude/settings.json .codex/config.toml .mcp.json +  .planning/workstreams/workflows/.verification-ledger.json > "$scope_local_hashes"

{
  for scope_commit in $(git rev-list --first-parent --no-merges "$scope_endpoint"..HEAD); do
    git diff-tree --no-commit-id --name-only -r "$scope_commit"
  done
  for scope_merge in $(git rev-list --first-parent --merges "$scope_endpoint"..HEAD); do
    git show --format= --name-only --remerge-diff "$scope_merge"
  done
} | LC_ALL=C sort -u > "$scope_manifest"

test -s "$scope_manifest"
git diff --no-ext-diff "$scope_endpoint"..HEAD -- $(cat "$scope_manifest") +  > "$scope_root/pr205-review.diff"
cat "$scope_manifest"
