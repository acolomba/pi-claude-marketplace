### `orchestrators/plugin/shared.ts` (verdict and selector)

**Analog:** `orchestrators/marketplace/shared.ts:1006-1061`. Copy this shape for `pluginLayerVerdict` and share one generic helper to stay under the fallow dupes threshold (3). The import direction is fixed: `plugin/shared.ts` already imports `../marketplace/shared.ts` (line 41); the reverse is forbidden, so the shared helper lives in the marketplace module or `persistence/`.

```ts
export type MarketplaceLayerVerdict =
  | { readonly kind: "declared"; readonly other: "declares" | "silent" | "unreadable" }
  | { readonly kind: "refused"; readonly reason: "marketplace declared locally" | "marketplace declared in shared config";
      readonly refusal: MarketplaceDeclaredLocallyError | MarketplaceDeclaredInSharedConfigError }
  | { readonly kind: "undeclared" }
  | { readonly kind: "other-unreadable"; readonly otherBasename: string };

export function marketplaceLayerVerdict(args: { target: ScopeConfig; other: ConfigLoadResult; targetIsLocal: boolean; marketplace: string }) {
  const inTarget = ownValue(args.target.marketplaces, args.marketplace) !== undefined;
  if (args.other.status === "invalid") {
    return inTarget
      ? { kind: "declared", other: "unreadable" }
      : { kind: "other-unreadable", otherBasename: path.basename(args.other.filePath) };
  }
  const inOther = args.other.status === "valid" &&
    ownValue(args.other.config.marketplaces, args.marketplace) !== undefined;
  if (inTarget) { return { kind: "declared", other: inOther ? "declares" : "silent" }; }
  if (!inOther) { return { kind: "undeclared" }; }
  return args.targetIsLocal
    ? { kind: "refused", reason: "marketplace declared in shared config", refusal: new MarketplaceDeclaredInSharedConfigError(args.marketplace) }
    : { kind: "refused", reason: "marketplace declared locally", refusal: new MarketplaceDeclaredLocallyError(args.marketplace) };
}
```

The retained return shape is the `DeclaringConfigWriteTarget` union (`:635-646`: `unreadable | selected {targetConfigPath, targetIsLocal, current, sibling}`). Make the other layer's invalid state an `unreadable` arm too (3.2 WR-03 rule, `remove.ts:624-629`).

**Own-key membership** (every read): `ownValue(cfg.plugins, key) !== undefined` (`shared/own-key.ts`), as in `layerRemovalEntries` (`remove.ts:667-676`).
